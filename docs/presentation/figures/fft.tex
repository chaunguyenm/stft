% Stage nodes
%
% #1 = x position
% #2 = overlay specification
% #3 = node prefix
% #4 = fill
%
\newcommand{\drawstage}[4]{%
    \foreach \i in {0,...,7} {
        \node<#2>[
            circle,
            fill=#4,
            inner sep=0,
            minimum size=.1cm
        ] (#3\i) at (#1,-\i*\rowsep) {};
    }
}

% Variable box
%
% #1 = overlay
% #2 = node name
% #3 = x
% #4 = y
% #5 = label
% #6 = number of parts
% #7 = fill specification
% #8 = contents
%
\newcommand{\varbox}[8]{%
    \node<#1>[
        rectangle split,
        rectangle split horizontal,
        rectangle split parts=#6,
        draw,
        rectangle split part fill={#7},
        label=left:{#5}
    ] (#2) at (#3,#4) {#8};
}
